= Getting started

== Maven dependency

[source,xml]
----
<dependency>
  <groupId>io.vidocq.knock</groupId>
  <artifactId>knock-cassini</artifactId>
  <version>0.1.0-SNAPSHOT</version>
</dependency>
----

== First check

[source,java]
----
@Liveness
@ApplicationScoped
public class DatabaseCheck implements HealthCheck {
    @Override
    public HealthCheckResponse call() {
        return HealthCheckResponse.named("database").up().build();
    }
}
----
